-- Modules/UIElementFactory.lua
local TweenService = game:GetService("TweenService")

local UIElementFactory = {}

function UIElementFactory:new()
    local obj = {TweenService = TweenService}
    setmetatable(obj, self)
    self.__index = self
    return obj
end

function UIElementFactory:CreateCorner(instance, radius)
    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, radius)
    corner.Parent = instance
    return corner
end

function UIElementFactory:CreateGradient(instance, colorPoints, rotation)
    local grad = Instance.new("UIGradient")
    grad.Color = ColorSequence.new(colorPoints)
    grad.Rotation = rotation or 0
    grad.Parent = instance
    return grad
end

function UIElementFactory:CreateStroke(instance, color, thickness, transparency)
    local stroke = Instance.new("UIStroke")
    stroke.Color = color
    stroke.Thickness = thickness
    stroke.Transparency = transparency or 0
    stroke.Parent = instance
    return stroke
end

function UIElementFactory:CreateTextLabel(parent, data)
    local label = Instance.new("TextLabel")
    label.Text = data.text
    label.Size = data.size
    label.Position = data.position
    label.BackgroundTransparency = data.bgTransparency or 1
    label.TextColor3 = data.color
    label.Font = data.font or Enum.Font.Gotham
    label.TextSize = data.fontSize
    label.TextXAlignment = data.xAlign or Enum.TextXAlignment.Center
    label.TextYAlignment = data.yAlign or Enum.TextYAlignment.Center
    label.ZIndex = data.zIndex or 5
    label.Parent = parent
    
    if data.wrapped then
        label.TextWrapped = true
    end
    if data.lineHeight then
        label.LineHeight = data.lineHeight
    end
    
    return label
end

function UIElementFactory:CreateButton(parent, data)
    local button = Instance.new("TextButton")
    button.Text = data.text
    button.Size = data.size
    button.Position = data.position
    button.BackgroundColor3 = data.bgColor
    button.TextColor3 = data.textColor
    button.Font = data.font or Enum.Font.GothamBold
    button.TextSize = data.fontSize or 15
    button.BorderSizePixel = 0
    button.ZIndex = data.zIndex or 5
    button.Parent = parent
    
    if data.cornerRadius then
        self:CreateCorner(button, data.cornerRadius)
    end
    if data.stroke then
        self:CreateStroke(button, data.stroke.color, data.stroke.thickness, data.stroke.transparency)
    end
    
    return button
end

function UIElementFactory:CreateFrame(parent, data)
    local frame = Instance.new("Frame")
    frame.Size = data.size
    frame.Position = data.position
    frame.BackgroundColor3 = data.bgColor or Color3.new(1, 1, 1)
    frame.BackgroundTransparency = data.bgTransparency or 0
    frame.BorderSizePixel = data.borderSize or 0
    frame.ZIndex = data.zIndex or 1
    frame.Parent = parent
    
    if data.cornerRadius then
        self:CreateCorner(frame, data.cornerRadius)
    end
    if data.gradient then
        self:CreateGradient(frame, data.gradient.colors, data.gradient.rotation)
    end
    if data.stroke then
        self:CreateStroke(frame, data.stroke.color, data.stroke.thickness, data.stroke.transparency)
    end
    
    return frame
end

-- Modules/AnimationController.lua
local AnimationController = {}
AnimationController.__index = AnimationController

function AnimationController:new(tweenService)
    local obj = {
        TweenService = tweenService,
        activeTweens = {}
    }
    setmetatable(obj, self)
    return obj
end

function AnimationController:CreateTween(target, properties, duration, easingStyle, easingDirection)
    local tweenInfo = TweenInfo.new(duration, easingStyle or Enum.EasingStyle.Quad, 
                                    easingDirection or Enum.EasingDirection.Out)
    local tween = self.TweenService:Create(target, tweenInfo, properties)
    table.insert(self.activeTweens, tween)
    tween:Play()
    return tween
end

function AnimationController:PlayEntrance(backdrop, blur, card, config)
    self:CreateTween(backdrop, {BackgroundTransparency = 0.45}, 0.35)
    self:CreateTween(blur, {Size = config.BlurSize}, config.AnimationDuration)
    task.wait(0.1)
    self:CreateTween(card, {
        Position = UDim2.new(0.5, -180, 0.5, -160),
        BackgroundTransparency = 0.2
    }, config.AnimationDuration, Enum.EasingStyle.Back, Enum.EasingDirection.Out)
end

function AnimationController:ButtonHover(button, hoverColor, originalColor, stroke)
    self:CreateTween(button, {BackgroundColor3 = hoverColor}, 0.15)
    if stroke then
        self:CreateTween(stroke, {Transparency = 0.2}, 0.15)
    end
end

function AnimationController:ButtonLeave(button, originalColor, stroke)
    self:CreateTween(button, {BackgroundColor3 = originalColor}, 0.15)
    if stroke then
        self:CreateTween(stroke, {Transparency = 0.55}, 0.15)
    end
end

function AnimationController:ButtonClick(button)
    self:CreateTween(button, {Size = UDim2.new(0.79, 0, 0, 42)}, 0.07)
    task.wait(0.07)
    self:CreateTween(button, {Size = UDim2.new(0.82, 0, 0, 46)}, 0.12, Enum.EasingStyle.Back)
end

function GroupAccessUI:CheckMembership()

end

function GroupAccessUI:CreateCopyButton()

        text = "🔗  Copy Group Link",
        size = UDim2.new(0.82, 0, 0, 46),
        position = UDim2.new(0.09, 0, 0, 262),
        bgColor = self.Config.Colors.Primary,
        textColor = self.Config.Colors.Text,
        zIndex = 5,
        cornerRadius = 12,
        stroke = {
            color = Color3.fromRGB(255, 255, 255),
            thickness = 1.5,
            transparency = 0.55
        }
    })
    
    self:SetupButtonEvents()
end

function GroupAccessUI:SetupButtonEvents()
    local originalColor = self.Config.Colors.Primary
    local hoverColor = self.Config.Colors.PrimaryHover
    local stroke = self.CopyButton:FindFirstChild("UIStroke")
    
    self.CopyButton.MouseEnter:Connect(function()
        self.AnimController:ButtonHover(self.CopyButton, hoverColor, originalColor, stroke)
    end)
    
    self.CopyButton.MouseLeave:Connect(function()
        self.AnimController:ButtonLeave(self.CopyButton, originalColor, stroke)
    end)
    
    self.CopyButton.MouseButton1Click:Connect(function()
        self.AnimController:ButtonClick(self.CopyButton)
        
        if setclipboard then
            setclipboard(self.Config.Group.Link)
            self.CopyButton.Text = "✓  Link Copied!"
            self.CopyButton.TextColor3 = self.Config.Colors.Success
            task.delay(2.5, function()
                self.CopyButton.Text = "🔗  Copy Group Link"
                self.CopyButton.TextColor3 = self.Config.Colors.Text
            end)
        else
            self.CopyButton.Text = "Not supported ⚠"
        end
    end)
end

function GroupAccessUI:Show()
    self:CreateBase()
    self:CreateCard()
    self.AnimController:PlayEntrance(self.Backdrop, self.Blur, self.Card, self.Config.Visuals)
end

local Players = game:GetService("Players")

local function main()

    if player:IsInGroup(UIConfig.Group.Id) then
